// ---------------------------------------------------------------- recorder
interface Row { scn: string; label: string; hand: unknown; app: unknown; ok: boolean; known?: string }
const ROWS: Row[] = []
function chk(scn: string, label: string, hand: unknown, app: unknown, known?: string): void {
  ROWS.push({ scn, label, hand, app, ok: JSON.stringify(hand) === JSON.stringify(app), ...(known ? { known } : {}) })
}
function assertScenario(scn: string): void {
  const bad = ROWS.filter((r) => r.scn === scn && !r.ok && !r.known)
  expect(bad).toEqual([])
}

// ---------------------------------------------------------------- company
interface Co { db: DB; info: CompanyInfo; L: (n: string) => number; I: (n: string) => number }
function company(info: CompanyInfo): Co {
  const db = seededDb()
  writeCompanyInfo(db, info)
  const gid = (name: string): number => (db.prepare('SELECT id FROM groups WHERE name = ?').get(name) as { id: number }).id
  const led = (name: string, group: string, extra: Record<string, unknown> = {}): void => {
    createLedger(db, { name, groupId: gid(group), ...extra } as Parameters<typeof createLedger>[1])
  }
  led('Sales', 'Sales Accounts')
  led('Purchases', 'Purchase Accounts')
  for (const t of ['cgst', 'sgst', 'igst'] as const) {
    led(`Output ${t.toUpperCase()}`, 'Duties & Taxes', { taxType: t })
    led(`Input ${t.toUpperCase()}`, 'Duties & Taxes', { taxType: t })
  }
  led('Round Off', 'Indirect Expenses')
  led('HDFC Bank', 'Bank Accounts')
  led('Staff Welfare', 'Indirect Expenses', { gstRate: 18, hsn: '996334' })
  led('Freight Inward', 'Direct Expenses', { gstRate: 5, hsn: '996511' })
  led('Umbrella Retail', 'Sundry Debtors', { stateCode: '27', gstin: '27AABCD1234E1Z8' })
  led('Krishna Enterprises', 'Sundry Debtors', { stateCode: '29', gstin: '29AABCF9012G1ZQ' })
  led('Suresh Rao', 'Sundry Debtors', { stateCode: '29' })
  led('Chennai Walk-in', 'Sundry Debtors', { stateCode: '33' })
  led('Delhi SEZ Unit', 'Sundry Debtors', { stateCode: '07', gstin: '07AABCK4444Q1Z4', exportType: 'sez_wop' })
  led('Global Handel GmbH', 'Sundry Debtors', { stateCode: '96', exportType: 'exp_wop' })
  led('Prepaid Clinic', 'Sundry Debtors', { stateCode: '27', gstin: '27AABCR2222N1ZD', gstRate: 18 })
  led('Bharat Steel', 'Sundry Creditors', { stateCode: '27', gstin: '27AABCG3456H1ZN' })
  led('Sharma Transport', 'Sundry Creditors', { stateCode: '27', rcm: true })
  const unit = (db.prepare("SELECT id FROM units WHERE symbol = 'Nos'").get() as { id: number }).id
  // Widget: 1,000 Nos at ₹100 = ₹1,00,000 (10,000,000 p). Rice: 500 Nos at ₹30 = ₹15,000 (1,500,000 p).
  createStockItem(db, stockItemInputSchema.parse({ name: 'Widget', unitId: unit, gstRate: 18, hsn: '8471', openingQtyMilli: 1_000_000, openingValue: 10_000_000 }))
  createStockItem(db, stockItemInputSchema.parse({ name: 'Rice', unitId: unit, gstRate: 5, hsn: '1006', openingQtyMilli: 500_000, openingValue: 1_500_000 }))
  const L = (n: string): number => (db.prepare('SELECT id FROM ledgers WHERE name = ?').get(n) as { id: number }).id
  const I = (n: string): number => (db.prepare('SELECT id FROM stock_items WHERE name = ?').get(n) as { id: number }).id
  return { db, info, L, I }
}

let seq = 0
const post = (c: Co, intent: Record<string, unknown>) => postIntent(c.db, c.info, { clientId: `readiness-${++seq}-x`, ...intent }, TODAY)
const typeId = (db: DB, kind: VoucherKind): number => (db.prepare('SELECT id FROM voucher_types WHERE kind = ? ORDER BY id LIMIT 1').get(kind) as { id: number }).id
const p = (rupees: number | undefined): number => Math.round((rupees ?? 0) * 100)
/** TB balance dr-positive, paise. */
function tb(c: Co, asOn: string): (name: string) => number {
  const t = trialBalance(c.db, asOn)
  return (name) => {
    const r = t.rows.find((x) => x.ledgerName === name)
    return r ? r.debit - r.credit : 0
  }
}
function lineOf(c: Co, voucherId: number, name: string): number {
  const v = getVoucher(c.db, voucherId)!
  return v.lines.filter((l) => l.ledgerId === c.L(name)).reduce((s, l) => s + (l.drCr === 'dr' ? l.amount : -l.amount), 0)
}
const lineUid = (c: Co, voucherId: number, item: string): string =>
  (c.db.prepare('SELECT line_uid AS u FROM inventory_lines WHERE voucher_id = ? AND stock_item_id = ?').get(voucherId, c.I(item)) as { u: string }).u
type J = Record<string, any>
const j1 = (c: Co, m: typeof OCT): J => gstr1(c.db, c.info, m.from, m.to, m.p).json as J
const j3 = (c: Co, m: typeof OCT): J => gstr3b(c.db, c.info, m.from, m.to, m.p).json as J
const tax3 = (o: J | undefined) => ({ txval: p(o?.txval), iamt: p(o?.iamt), camt: p(o?.camt), samt: p(o?.samt) })
const itc3 = (o: J | undefined) => ({ iamt: p(o?.iamt), camt: p(o?.camt), samt: p(o?.samt) })
const itcRow = (j: J, part: 'itc_avl' | 'itc_rev', ty: string) => itc3((j.itc_elg[part] as J[]).find((r) => r.ty === ty))
const itms = (list: J[] | undefined) =>
  (list ?? []).map((i) => i.itm_det ?? i).map((d: J) => ({ rt: d.rt, txval: p(d.txval), iamt: p(d.iamt), camt: p(d.camt), samt: p(d.samt) })).sort((a, b) => a.rt - b.rt)
const owedOf = (c: Co, side: 'receivable' | 'payable', name: string): number =>
  outstandingsWithAdvances(c.db, side, TODAY).parties.find((x) => x.name === name)?.pending ?? 0
const home = (c: Co) => simpleHome(c.db, c.info, { today: TODAY, backups: [] })
const taxes = (c: Co) => simpleTaxes(c.db, c.info, {}, TODAY)
