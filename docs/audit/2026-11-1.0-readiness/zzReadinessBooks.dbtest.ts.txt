const TODAY = '2026-11-18'

let seq = 0

function shop(gst: 'regular' | 'composition' | 'no', opts: { stock?: boolean; type?: 'trader' | 'manufacturer'; start?: string; today?: string } = {}): { db: DB; info: CompanyInfo } {
  const db = freshDb()
  applySetup(db, {
    name: `Audit ${gst}`, stateCode: '27', businessType: opts.type ?? 'trader', gst, gstin: gst === 'no' ? null : GSTIN,
    filing: gst === 'regular' ? 'monthly' : null, stock: opts.stock ?? true, startDate: opts.start ?? '2026-04-01', fresh: false
  }, opts.today ?? TODAY)
  return { db, info: readCompanyInfo(db) }
}
const gid = (db: DB, name: string): number => (db.prepare('SELECT id FROM groups WHERE name = ?').get(name) as { id: number }).id
const lid = (db: DB, name: string): number => (db.prepare('SELECT id FROM ledgers WHERE name = ?').get(name) as { id: number }).id
const unit = (db: DB, symbol: string): number => (db.prepare('SELECT id FROM units WHERE symbol = ?').get(symbol) as { id: number }).id
const vt = (db: DB, kind: string): number => (db.prepare('SELECT id FROM voucher_types WHERE kind = ? ORDER BY id').get(kind) as { id: number }).id
const sec = (db: DB, code: string): number => (db.prepare('SELECT id FROM tds_sections WHERE code = ?').get(code) as { id: number }).id
function ledger(db: DB, name: string, group: string, extra: Record<string, unknown> = {}): number {
  return createLedger(db, ledgerInputSchema.parse({ name, groupId: gid(db, group), stateCode: '27', ...extra })).id
}
function item(db: DB, name: string, extra: Record<string, unknown> = {}): number {
  return createStockItem(db, stockItemInputSchema.parse({ name, unitId: unit(db, 'Nos'), gstRate: 18, ...extra })).id
}
const post = (db: DB, info: CompanyInfo, intent: object, today = TODAY) => postIntent(db, info, { clientId: `audit-${++seq}-xyz`, ...intent }, today)
/** Signed (dr-positive) TB balance of one ledger. */
function tbBal(db: DB, asOn: string, name: string): number {
  const r = trialBalance(db, asOn).rows.find((x) => x.ledgerName === name)
  return r ? r.debit - r.credit : 0
}
function tbByTax(db: DB, asOn: string, taxType: string, side: 'Input' | 'Output'): number {
  const ids = (db.prepare('SELECT id FROM ledgers WHERE tax_type = ? AND name LIKE ?').all(taxType, `%${side}%`) as { id: number }[]).map((r) => r.id)
  return trialBalance(db, asOn).rows.filter((r) => ids.includes(r.ledgerId)).reduce((s, r) => s + r.debit - r.credit, 0)
}

// 1. Openings — books begin 2026-04-01 (regular trader).
//   cash 10,000.00 = 10_000_00; bank 50,000.00 = 50_000_00;
//   customer Sharma 11,800.00 = bill B-1 7,000.00 dated 2026-02-20 (40 days before 1 Apr:
//   Feb 20→Mar 1 = 9 days (Feb 2026 has 28), Mar 1→Apr 1 = 31 → 40) + an undated bill 4,800.00
//   (→ "Opening", dated the books-begin day);
//   supplier Gupta 5,000.00; stock Widget 10 × 250.00 = 2,500.00.
//   have = 10,000 + 50,000 + 11,800 + 2,500 = 74,300.00; owe = 5,000.00;
//   capital (owner's money) = −(74,300 − 5,000) = −69,300.00 (a credit).
//   TB as on 2026-04-01: Dr 74,300.00 (cash, bank, Sharma, opening stock) = Cr 74,300.00 (capital
//   69,300 + Gupta 5,000).  BS: assets 74,300.00 = liabilities 74,300.00.
//   Outstandings as on 2026-05-01: B-1 age = Feb 20→May 1 = 8 (rest of Feb) + 31 + 30 + 1 = 70 days;
//   no credit days → due = bill date → overdue 70 → bucket 61-90. "Opening" age Apr 1→May 1 = 30
//   → bucket 0-30.  buckets = [4,800.00, 0, 7,000.00, 0].
// 5. (same company, continuing) Stock count on 2026-05-10: Widget counted 8 (2 short at the 250.00
//   average → −500.00), Jam (never stocked) counted 5 at a stated cost of 40.00 → +200.00.
//   Closing stock = 8 × 250 + 5 × 40 = 2,000 + 200 = 2,200.00.
//   P&L 2026-04-01..2026-05-31: opening stock 2,500.00, purchases 0, closing 2,200.00 →
//   cost of goods 300.00, sales 0 → profit −300.00.  BS on 2026-05-31: assets 74,300 − 300 =
//   74,000.00; liabilities 69,300 + 5,000 − 300 = 74,000.00.
// ------------------------------------------------------------------------------------------------
describe('1 + 5. openings, then a stock count', () => {
  it('openings: capital difference, TB, BS, Outstandings ageing; count: stock, P&L, BS, simple screens', () => {
    const { db, info } = shop('regular')
    const input = {
      cash: { amount: 10_000_00, base: 0 },
      banks: [{ ledgerId: null, name: 'HDFC', amount: 50_000_00, base: null }],
      customers: [{ ledgerId: null, name: 'Sharma', base: null, bills: [{ ref: 'B-1', date: '2026-02-20', amount: 7_000_00 }, { ref: '', date: null, amount: 4_800_00 }] }],
      suppliers: [{ ledgerId: null, name: 'Gupta', base: null, bills: [{ ref: 'G-1', date: null, amount: 5_000_00 }] }],
      items: [{ itemId: null, name: 'Widget', unitId: null, qtyMilli: 10_000, ratePaise: 250_00, base: null, gstRate: 18 }],
      others: []
    }
    const p = previewOpenings(db, input)
    expect(p.capital.after).toBe(-69_300_00)
    applyOpenings(db, { ...input, confirmCapital: p.capital.after })
    const tb = trialBalance(db, '2026-04-01')
    expect(tb.totalDebit).toBe(74_300_00)
    expect(tb.totalCredit).toBe(74_300_00)
    const bs = balanceSheet(db, '2026-04-01', '2026-04-01')
    expect(bs.totalAssets).toBe(74_300_00)
    expect(bs.totalLiabilities).toBe(74_300_00)
    const rec = outstandingsWithAdvances(db, 'receivable', '2026-05-01').parties.find((x) => x.name === 'Sharma')!
    expect(rec.pending).toBe(11_800_00)
    expect(rec.bills.map((b) => [b.number, b.date, b.ageDays, b.overdueDays, b.pending])).toEqual([
      ['B-1', '2026-02-20', 70, 70, 7_000_00],
      ['Opening', '2026-04-01', 30, 30, 4_800_00]
    ])
    expect(rec.buckets).toEqual([4_800_00, 0, 7_000_00, 0])
    const pay = outstandingsWithAdvances(db, 'payable', '2026-05-01').parties.find((x) => x.name === 'Gupta')!
    expect(pay.pending).toBe(5_000_00)
    // Simple Home as on 2026-05-01: customers owe 11,800, you owe 5,000, cash 10,000 + bank 50,000.
    const home = simpleHome(db, info, { today: '2026-05-01', backups: [] })
    expect(home.receivables.total).toBe(11_800_00)
    expect(home.payables.total).toBe(5_000_00)
    expect(home.cash.cash).toBe(10_000_00)
    expect(home.cash.bank).toBe(50_000_00)
    // People: opening bills without a due date are due on their own date → both late; the
    // oldest 70 days.
    const sharma = peopleList(db, '2026-05-01').people.find((x) => x.name === 'Sharma')!
    expect([sharma.open, sharma.late, sharma.lateBills, sharma.maxLateDays]).toEqual([11_800_00, 11_800_00, 2, 70])
    // Stock list = 10 × 250 = 2,500.
    expect(stockList(db, '2026-05-01').rows.find((r) => r.name === 'Widget')).toMatchObject({ qtyMilli: 10_000, value: 2_500_00 })

    // ---- 5. the count
    const widget = (db.prepare("SELECT id FROM stock_items WHERE name = 'Widget'").get() as { id: number }).id
    const jam = item(db, 'Jam')
    const out = post(db, info, { kind: 'stock_count', date: '2026-05-10', counts: [{ itemId: widget, qty: '8' }, { itemId: jam, qty: '5', cost: '40' }] }, '2026-05-15')
    expect(out.facts.count!.lines.map((l) => [l.beforeMilli, l.countedMilli, l.valueChange])).toEqual([[10_000, 8_000, -500_00], [0, 5_000, 200_00]])
    const list = stockList(db, '2026-05-31').rows
    expect(list.find((r) => r.itemId === widget)).toMatchObject({ qtyMilli: 8_000, value: 2_000_00 })
    expect(list.find((r) => r.itemId === jam)).toMatchObject({ qtyMilli: 5_000, value: 200_00 })
    const ss = stockSummary(db, '2026-05-31')
    expect(ss.reduce((s, r) => s + r.closingValue, 0)).toBe(2_200_00)
    const pl = profitAndLoss(db, '2026-04-01', '2026-05-31')
    expect([pl.openingStock, pl.closingStock, pl.grossProfit, pl.netProfit]).toEqual([2_500_00, 2_200_00, -300_00, -300_00])
    const doing = businessDoing(db, { period: 'custom', from: '2026-04-01', to: '2026-05-31' }, '2026-06-15')
    expect([doing.lines.costOfGoods, doing.lines.profit]).toEqual([300_00, -300_00])
    const bs2 = balanceSheet(db, '2026-04-01', '2026-05-31')
    expect([bs2.totalAssets, bs2.totalLiabilities, bs2.profitCurrentPeriod]).toEqual([74_000_00, 74_000_00, -300_00])
    const tb2 = trialBalance(db, '2026-05-31')
    expect(tb2.totalDebit).toBe(tb2.totalCredit)
  })
})

    const bala = ledger(db, 'Bala', 'Sundry Debtors')

// 4. Stock: Widget (weighted average — the default, stock_items.valuation_method) and Gadget (FIFO).
//   Each: buy 10 @ 100.00 (2026-10-01), buy 10 @ 130.00 (2026-10-05), sell 15 @ 200.00 (2026-10-10).
//   Widget WA: 20 units 1,000 + 1,300 = 2,300.00 → avg 115.00; COGS 15 × 115 = 1,725.00;
//     closing 5 × 115 = 575.00.
//   Gadget FIFO: COGS 10 × 100 + 5 × 130 = 1,650.00; closing 5 × 130 = 650.00.
//   Purchases 2 × 2,300 = 4,600.00 (+ input GST 18 % = 828.00: CGST 414.00 + SGST 414.00).
//   Sales 2 × 3,000 = 6,000.00 (+ output GST 1,080.00). Closing stock 575 + 650 = 1,225.00.
//   Cost of goods = 0 + 4,600 − 1,225 = 3,375.00 (= 1,725 + 1,650). Gross = net = 6,000 − 3,375 =
//   2,625.00. BS: assets stock 1,225 + customer 7,080 + input GST 828 = 9,133.00; liabilities
//   supplier 5,428 + output GST 1,080 + profit 2,625 = 9,133.00.
//   Purchase bills: 2,000 + 360 = 2,360.00; 2,600 + 468 = 3,068.00 → supplier 5,428.00.
// ------------------------------------------------------------------------------------------------
describe('4. stock: weighted average and FIFO', () => {
  it('cost of goods, closing stock, gross profit in P&L, simple doing and stock list', () => {
    const { db, info } = shop('regular')
    const sup = ledger(db, 'Bharat', 'Sundry Creditors')
    const cus = ledger(db, 'Kiran', 'Sundry Debtors')
    const w = item(db, 'Widget')
    const g = item(db, 'Gadget', { valuationMethod: 'fifo' })
    post(db, info, { kind: 'purchase', supplierId: sup, billNo: 'P-1', date: '2026-10-01', items: [{ itemId: w, qty: '10', rate: '100' }, { itemId: g, qty: '10', rate: '100' }], payment: { how: 'credit' } })
    post(db, info, { kind: 'purchase', supplierId: sup, billNo: 'P-2', date: '2026-10-05', items: [{ itemId: w, qty: '10', rate: '130' }, { itemId: g, qty: '10', rate: '130' }], payment: { how: 'credit' } })
    post(db, info, { kind: 'sale', customerId: cus, date: '2026-10-10', items: [{ itemId: w, qty: '15', rate: '200' }, { itemId: g, qty: '15', rate: '200' }], payment: { how: 'credit' } })
    const list = stockList(db, TODAY).rows

    expect(list.find((r) => r.itemId === g)).toMatchObject({ qtyMilli: 5_000, value: 650_00 })
    const pl = profitAndLoss(db, '2026-04-01', TODAY)
    expect([pl.openingStock, pl.closingStock, pl.grossProfit, pl.netProfit]).toEqual([0, 1_225_00, 2_625_00, 2_625_00])
    const d = businessDoing(db, { period: 'custom', from: '2026-04-01', to: TODAY }, TODAY)
    expect([d.lines.sales, d.lines.purchases, d.lines.closingStock, d.lines.costOfGoods, d.lines.profit]).toEqual([6_000_00, 4_600_00, 1_225_00, 3_375_00, 2_625_00])
    expect(tbBal(db, TODAY, 'Bharat')).toBe(-5_428_00)
    expect(tbBal(db, TODAY, 'Kiran')).toBe(7_080_00)
    expect(tbByTax(db, TODAY, 'cgst', 'Input') + tbByTax(db, TODAY, 'sgst', 'Input')).toBe(828_00)
    const bs = balanceSheet(db, '2026-04-01', TODAY)
    // Re-checked: the BS shows Duties & Taxes as one net group — output 1,080 − input 828 = 252
      // liability — so assets 1,225 + 7,080 = 8,305.00 = 5,428 + 252 + 2,625 (Tally presents the
      // group net the same way; my first hand figure grossed input GST up as an asset).
      expect([bs.totalAssets, bs.totalLiabilities, bs.profitCurrentPeriod]).toEqual([8_305_00, 8_305_00, 2_625_00])
    expect(purchasesList(db, '2026-04-01', TODAY, TODAY).map((r) => r.total).sort()).toEqual([2_360_00, 3_068_00])
    const home = simpleHome(db, info, { today: TODAY, backups: [] })
    expect(home.profit.year.netProfit).toBe(2_625_00)
  })
})

describe('6. made goods', () => {

// 7. TDS 194C, FY 2025-26 (seeded company, books from 2025). Rate rows (migration 020): 1 %
//   individual / HUF, 2 % others; single > 30,000.00 OR FY aggregate > 1,00,000.00
//   (s.194C(5) and its proviso: "does not exceed thirty thousand rupees … aggregate … exceeds one
//   lakh rupees").
//  7a. Four payments of 25,000.00 to contractor P (PAN 4th letter P = individual), no bills
//   (advance events): aggregate 1,00,000.00 — NOT more than 1,00,000 → nothing eligible.
//   A fifth 25,000.00 → 1,25,000.00 > 1,00,000 → all five liable: 5 × 25,000 × 1 % = 5 × 250.00
//   = 1,250.00.
//  7b. Bills (journals) to contractor B: 30,000 (May) + 30,000 (Jun) + 30,000 (Jul) = 90,000;
//   debit note 15 Jul returns 10,000 → aggregate 80,000; bill 20,000 (Aug) → 1,00,000 — not
//   exceeded → nothing eligible. Twin C without the return: 1,10,000 > 1,00,000 → all four
//   liable, 1,10,000 × 1 % = 1,100.00.
//  7c. Return AFTER crossing (contractor D = C's bills, crossed at 1,10,000 in Aug), DN 10,000
//   on 15 Aug lowers nothing; bill 5,000 on 1 Sep still liable: 5,000 × 1 % = 50.00.
//  7d. Company contractor E (PAN 4th letter C): one bill 40,000.00 (single > 30,000) → 2 % =
//   800.00; deducted on the bill: Dr Exp 40,000 / Cr E 39,200 / Cr TDS payable 800.00.
// ------------------------------------------------------------------------------------------------
describe('7. TDS 194C', () => {
  const contractor = (db: DB, name: string, pan: string): number => ledger(db, name, 'Sundry Creditors', { tdsSectionId: sec(db, '194C'), pan, stateCode: null })
  const expense = (db: DB, name = 'Contract work'): number => {
    const r = db.prepare('SELECT id FROM ledgers WHERE name = ?').get(name) as { id: number } | undefined
    return r?.id ?? ledger(db, name, 'Indirect Expenses', { stateCode: null })
  }
  const base = { partyLedgerId: null, narration: null, reference: null, billRefs: [], inventory: [] }
  const pay = (db: DB, party: number, date: string, amt: number) => saveVoucher(db, { ...base, voucherTypeId: vt(db, 'payment'), date, partyLedgerId: party, lines: [{ ledgerId: party, drCr: 'dr', amount: amt }, { ledgerId: lid(db, 'Cash'), drCr: 'cr', amount: amt }], tds: null })
  const bill = (db: DB, party: number, date: string, amt: number) => saveVoucher(db, { ...base, voucherTypeId: vt(db, 'journal'), date, partyLedgerId: party, lines: [{ ledgerId: expense(db), drCr: 'dr', amount: amt }, { ledgerId: party, drCr: 'cr', amount: amt }], tds: null })
  const dn = (db: DB, party: number, date: string, amt: number) => saveVoucher(db, { ...base, voucherTypeId: vt(db, 'debit_note'), date, partyLedgerId: party, lines: [{ ledgerId: party, drCr: 'dr', amount: amt }, { ledgerId: expense(db), drCr: 'cr', amount: amt }], tds: null })
  const eligible = (db: DB, party: number) => tdsEligible(db, '2025-04-01', '2026-03-31').filter((r) => r.partyLedgerId === party)

  it('7a. four payments of 25,000 → none; the fifth → all five at 1 %', () => {
    const db = seededDb()
    const p = contractor(db, 'Contractor P', 'ABCPE1234F')
    for (const d of ['2025-05-01', '2025-06-01', '2025-07-01', '2025-08-01']) pay(db, p, d, 25_000_00)
    expect(eligible(db, p)).toEqual([])
    pay(db, p, '2025-09-01', 25_000_00)
    const rows = eligible(db, p)
    expect(rows.map((r) => [r.basePaise, r.rateBp, r.tdsPaise])).toEqual(Array(5).fill([25_000_00, 100, 250_00]))
    expect(rows.reduce((s, r) => s + (r.tdsPaise ?? 0), 0)).toBe(1_250_00)
  })

  it('7b. a purchase return before the crossing keeps the aggregate at 1,00,000 → none; the twin crosses', () => {
    const db = seededDb()
    const b = contractor(db, 'Contractor B', 'ABCPE1234F')
    const c = contractor(db, 'Contractor C', 'ABCPE1235F')
    for (const x of [b, c]) {
      bill(db, x, '2025-05-01', 30_000_00)
      bill(db, x, '2025-06-01', 30_000_00)
      bill(db, x, '2025-07-01', 30_000_00)
    }
    dn(db, b, '2025-07-15', 10_000_00)
    for (const x of [b, c]) bill(db, x, '2025-08-01', 20_000_00)
    expect(eligible(db, b)).toEqual([])
    const rows = eligible(db, c)
    expect(rows.map((r) => r.basePaise)).toEqual([30_000_00, 30_000_00, 30_000_00, 20_000_00])
    expect(rows.reduce((s, r) => s + (r.tdsPaise ?? 0), 0)).toBe(1_100_00)
  })

  it('7c. a return after the crossing lowers nothing: a later 5,000 bill stays liable (50.00)', () => {
    const db = seededDb()
    const d = contractor(db, 'Contractor D', 'ABCPE1234F')
    bill(db, d, '2025-05-01', 30_000_00)
    bill(db, d, '2025-06-01', 30_000_00)
    bill(db, d, '2025-07-01', 30_000_00)
    bill(db, d, '2025-08-01', 20_000_00)
    dn(db, d, '2025-08-15', 10_000_00)
    const late = bill(db, d, '2025-09-01', 5_000_00)
    const r = eligible(db, d).find((x) => x.voucherId === late.id)!
    expect([r.basePaise, r.tdsPaise]).toEqual([5_000_00, 50_00])
  })

  it('7d. a company contractor, single bill 40,000 → 2 % = 800, posted to the TDS payable ledger', () => {
    const db = seededDb()
    const e = contractor(db, 'Contractor E', 'ABCCE1234F')
    const s = tdsSuggestion(db, e, 40_000_00, '2025-05-01', { voucherKind: 'journal' })!
    expect([s.rateBp, s.tdsPaise, s.thresholdCrossed]).toEqual([200, 800_00, true])
    saveVoucher(db, {
      ...base, voucherTypeId: vt(db, 'journal'), date: '2025-05-01', partyLedgerId: e,
      lines: [{ ledgerId: expense(db), drCr: 'dr', amount: 40_000_00 }, { ledgerId: e, drCr: 'cr', amount: 39_200_00 }],
      tds: { sectionId: sec(db, '194C'), baseAmount: 40_000_00, tdsAmount: 800_00, isManual: false, autoPayable: true }
    })
    const payable = findPayableLedger(db, sec(db, '194C'))!
    const tb = trialBalance(db, '2025-05-31')
    const bal = (id: number): number => { const r = tb.rows.find((x) => x.ledgerId === id); return r ? r.debit - r.credit : 0 }
    expect([bal(payable.id), bal(e), bal(expense(db))]).toEqual([-800_00, -39_200_00, 40_000_00])
    expect(tb.totalDebit).toBe(tb.totalCredit)
    expect(eligible(db, e)).toEqual([])
  })

  it('7e. simple Money out preview: the fifth 25,000 payment shows a TDS hint', () => {
    const db = seededDb()
    const p = contractor(db, 'Contractor P', 'ABCPE1234F')
    for (const d of ['2025-05-01', '2025-06-01', '2025-07-01', '2025-08-01']) pay(db, p, d, 25_000_00)
    const info = readCompanyInfo(db)
    const pv = previewIntent(db, info, { clientId: 'audit-preview-1', kind: 'money_out', date: '2025-09-01', supplierId: p, amount: 25_000_00, from: { how: 'cash' } }, '2025-09-05')
    expect(pv.tds?.length ?? 0).toBeGreaterThan(0)
  })
})

// 10. Year boundary (seeded company, books from FY 2025-26).
//   FY 25-26: cash sale 10,000.00 (2025-06-01), rent paid 4,000.00 (2025-07-01) → profit 6,000.00.
//   Close FY 25-26 (closing journal): Sales Dr 10,000 / Rent Cr 4,000 / retained Cr 6,000.
//   P&L FY 25-26 still 6,000.00 (closing journals left out).
//   FY 26-27: cash sale 3,000.00 (2026-05-01) → P&L FY 26-27 = 3,000.00 (income reset); TB on
//   2026-05-31 shows Sales Cr 3,000.00 only. BS on 2026-05-31: cash 10,000 − 4,000 + 3,000 =
//   9,000.00 = retained 6,000 + current 3,000. A twin WITHOUT the close shows the same BS totals
//   and the same FY 26-27 profit (the brought-forward profit via yearOpening).

describe('10. year boundary', () => {
  function books(close: boolean): DB {
    const db = seededDb()
    const sales = ledger(db, 'Sales', 'Sales Accounts', { stateCode: null })
    const rent = ledger(db, 'Rent', 'Indirect Expenses', { stateCode: null })
    const cash = lid(db, 'Cash')

    v('receipt', '2026-05-01', cash, sales, 3_000_00)
    return db
  }
  it('closed and not closed: P&L per FY, TB reset, BS retained profit', () => {
